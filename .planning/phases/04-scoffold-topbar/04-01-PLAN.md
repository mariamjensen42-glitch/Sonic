```kotlin
package com.cycling.sonic.core.ui.components

import androidx.compose.foundation.layout.PaddingValues
import androidx.compose.material3.ExperimentalMaterial3Api
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Scaffold
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import com.cycling.sonic.core.ui.theme.SpotifyColors

/**
 * Unified layout container for screens using Scaffold + TopAppBar + content pattern.
 *
 * Wraps [Scaffold] with project-defaults (background color, content color)
 * and provides the standard topBar/content slot layout.
 *
 * Usage:
 * ```
 * M3Scaffold(
 *     topBar = { M3DetailTopAppBar(title = "标题", ...) }
 * ) { padding ->
 *     Box(Modifier.padding(padding)) { ... }
 * }
 * ```
 *
 * @param modifier Standard Compose modifier.
 * @param topBar Optional top bar slot (e.g., M3TopAppBar, M3DetailTopAppBar).
 * @param bottomBar Optional bottom bar slot.
 * @param containerColor Background color, defaults to SpotifyColors.Background.
 * @param contentColor Default content color, defaults to MaterialTheme.colorScheme.onBackground.
 * @param content Main content slot receiving Scaffold's contentPadding for inset-aware layout.
 */
@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun M3Scaffold(
    modifier: Modifier = Modifier,
    topBar: @Composable () -> Unit = {},
    bottomBar: @Composable () -> Unit = {},
    containerColor: Color = SpotifyColors.Background,
    contentColor: Color = MaterialTheme.colorScheme.onBackground,
    content: @Composable (PaddingValues) -> Unit
) {
    Scaffold(
        modifier = modifier,
        topBar = topBar,
        bottomBar = bottomBar,
        containerColor = containerColor,
        contentColor = contentColor,
        content = content
    )
}
```

Key design decisions (document in KDoc):
- `@OptIn(ExperimentalMaterial3Api::class)` stays INTERNAL to M3Scaffold — callers do NOT need it.
- `containerColor` defaults to `SpotifyColors.Background` (the project's dark theme background).
- All parameters pass directly to `material3.Scaffold` — no new semantics, just defaults.
- Keep `Modifier` as first optional param per Compose convention.
